import numpy as np
import matplotlib.pyplot as plt
----


+*In[2]:*+
[source, ipython3]
----
x = np.linspace(0, 1, 100)
----


+*In[3]:*+
[source, ipython3]
----
def sin_function(x):
    return np.sin(x)

def cos_function(x):
    return np.cos(x)
----


+*In[4]:*+
[source, ipython3]
----
y_sin = sin_function(x)
y_cos = cos_function(x)

fig, axes = plt.subplots(1, 2, figsize=(10, 4))

# Left panel: sin(x)
axes[0].plot(x, y_sin)
axes[0].set_title("sin(x)")
axes[0].set_xlabel("x")
axes[0].set_ylabel("sin(x)")

# Right panel: cos(x)
axes[1].plot(x, y_cos)
axes[1].set_title("cos(x)")
axes[1].set_xlabel("x")
axes[1].set_ylabel("cos(x)")

plt.tight_layout()
plt.savefig("trig_plots.pdf")
plt.show()
